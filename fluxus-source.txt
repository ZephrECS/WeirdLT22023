let currentHeroIndex = 0;
function startHeroRotation() {
    updateHero();
    setInterval(() => { currentHeroIndex = (currentHeroIndex + 1) % games.length; updateHero(); }, 8000);
}
function updateHero() {
    const g = games[currentHeroIndex];
    const bg = document.getElementById('heroBackground');
    const badge = document.getElementById('heroBadge');
    const title = document.getElementById('heroTitle');
    const desc = document.getElementById('heroDescription');
    const btns = document.querySelector('.hero-buttons');
    [badge, title, desc, btns].forEach(el => el.classList.remove('visible'));
    bg.classList.remove('active');
    setTimeout(() => {
        document.getElementById('heroImage').src = g.image;
        badge.textContent = g.tag || 'Featured Game';
        title.textContent = g.title;
        desc.textContent = g.description || '';
        bg.classList.add('active');
        setTimeout(() => badge.classList.add('visible'), 100);
        setTimeout(() => title.classList.add('visible'), 200);
        setTimeout(() => desc.classList.add('visible'), 300);
        setTimeout(() => btns.classList.add('visible'), 400);
    }, 600);
}

// ============================================
// RENDER GAMES
// ============================================
function renderGames() {
    const container = document.getElementById('allGamesGrid');

    const makeCard = (game, isSecret, idx) => {
        const div = document.createElement('div');
        div.className = 'bento-item' + (isSecret ? ' secret-game' : '');
        div.dataset.category = game.category || '';
        if (isSecret) div.style.display = 'none';

        div.innerHTML = `
            <img src="${game.image}" class="bento-image" alt="${game.title}">
            <div class="bento-overlay">
                <div class="bento-title">${game.title}</div>
                ${game.tag ? `<span class="bento-tag">${game.tag}</span>` : isSecret ? '<span class="bento-tag">SECRET</span>' : ''}
            </div>`;

        div.querySelector('.bento-image').addEventListener('load', function() {
            setGridRowSpan(this);
        });

        div.addEventListener('click', () => {
            if (game.versions) {
                showVersionPicker(idx, isSecret);
            } else {
                launchGame(game.url, game.title, game.loadMode || 'onsite');
            }
        });

        return div;
    };

    container.innerHTML = '';
    games.forEach((g, i) => container.appendChild(makeCard(g, false, i)));
    secretGames.forEach((g, i) => container.appendChild(makeCard(g, true, i)));

    updateCategoryCounts();
    setupCategoryFilters();
}

function setGridRowSpan(img) {
    const item = img.closest('.bento-item');
    const h = img.getBoundingClientRect().height;
    item.style.gridRowEnd = `span ${Math.ceil((h + 10) / 20)}`;
}

function updateCategoryCounts() {
    ['all','featured','action','horror','sandbox','classic'].forEach(cat => {
        const el = document.getElementById(`count-${cat}`);
        if (el) el.textContent = cat === 'all' ? games.length : games.filter(g => g.category === cat).length;
    });
}

function setupCategoryFilters() {
    document.querySelectorAll('.category-item').forEach(item => {
        item.addEventListener('click', function() {
            document.querySelectorAll('.category-item').forEach(i => i.classList.remove('active'));
            this.classList.add('active');
            filterByCategory(this.dataset.category);
        });
    });
}

function filterByCategory(category) {
    document.querySelectorAll('.bento-item:not(.secret-game)').forEach(item => {
        const show = category === 'all' || item.dataset.category === category;
        item.classList.toggle('fade-out', !show);
        item.style.visibility = show ? 'visible' : 'hidden';

    });
}

// ============================================
// LAUNCH GAME — routes by loadMode
// ============================================
async function launchGame(url, title, loadMode) {
    if (loadMode === 'newtab') {
        await openInNewTab(url);
    } else {
        await openOnsite(url, title);
    }
}

async function openInNewTab(url) {
    try {
        const res = await fetch(url);
        const code = await res.text();
        const tab = window.open('', '_blank');
        tab.document.open();
        tab.document.write(code);
        tab.document.close();
    } catch { alert('Error loading game'); }
}

// ============================================
let currentGameViewer = null;

async function openOnsite(url, title) {
    if (currentGameViewer) closeGameViewer(currentGameViewer, false);

    const viewer = document.createElement('div');
    viewer.className = 'game-viewer';
    viewer.innerHTML = `
        <div class="game-toolbar">
            <div class="game-toolbar-left">
                <div class="game-favicon">F</div>
                <div class="game-toolbar-title">${title || 'Game'}</div>
            </div>
            <div class="game-toolbar-right">
                <button class="gtb gtb-save" id="gv-save">
                    <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M19 21H5a2 2 0 0 1-2-2V5a2 2 0 0 1 2-2h11l5 5v11a2 2 0 0 1-2 2z"/><polyline points="17 21 17 13 7 13 7 21"/><polyline points="7 3 7 8 15 8"/></svg>
                    Save
                </button>
                <button class="gtb gtb-load" id="gv-load">
                    <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"/><polyline points="7 10 12 15 17 10"/><line x1="12" y1="15" x2="12" y2="3"/></svg>
                    Load Save
                </button>
                <div class="gtb-divider"></div>
                <button class="gtb gtb-refresh" id="gv-refresh">
                    <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M23 4v6h-6"/><path d="M20.49 15a9 9 0 1 1-2.12-9.36L23 10"/></svg>
                    Refresh
                </button>
                <button class="gtb gtb-fullscreen" id="gv-fullscreen">
                    <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M8 3H5a2 2 0 0 0-2 2v3m18 0V5a2 2 0 0 0-2-2h-3m0 18h3a2 2 0 0 0 2-2v-3M3 16v3a2 2 0 0 0 2 2h3"/></svg>
                    Fullscreen
                </button>
                <div class="gtb-divider"></div>
                <button class="gtb gtb-exit" id="gv-exit">
                    <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg>
                    Exit
                </button>
            </div>
        </div>
        <div class="game-iframe-wrap">
            <div class="game-loading">
                <div class="game-loading-text">Loading</div>
                <div class="game-loading-dots"><span></span><span></span><span></span></div>
            </div>
            <iframe id="gameFrame" allowfullscreen allow="autoplay; fullscreen; gamepad; keyboard-map; pointer-lock" style="opacity:0;transition:opacity 0.3s ease;"></iframe>
        </div>`;

    document.body.appendChild(viewer);
    currentGameViewer = viewer;

    const iframe = viewer.querySelector('#gameFrame');
    const loadingEl = viewer.querySelector('.game-loading');

    // Load via blob
    let blobUrl = null;
    try {
        const res = await fetch(url);
        const code = await res.text();
        const blob = new Blob([code], { type: 'text/html' });
        blobUrl = URL.createObjectURL(blob);
        iframe.addEventListener('load', () => {
            loadingEl.style.display = 'none';
            iframe.style.opacity = '1';
            setTimeout(() => URL.revokeObjectURL(blobUrl), 2000);
        }, { once: true });
        iframe.src = blobUrl;
    } catch {
        loadingEl.querySelector('.game-loading-text').textContent = 'Failed to load';
        loadingEl.querySelector('.game-loading-dots').style.display = 'none';
    }

    // ---- SAVE: export iframe's localStorage to .json ----
    viewer.querySelector('#gv-save').addEventListener('click', () => {
        try {
            // Attempt to read iframe localStorage (same-origin blob works)
            const iframeWin = iframe.contentWindow;
            const data = {};
            for (let i = 0; i < iframeWin.localStorage.length; i++) {
                const k = iframeWin.localStorage.key(i);
                data[k] = iframeWin.localStorage.getItem(k);
            }
            // Also grab sessionStorage
            const session = {};
            for (let i = 0; i < iframeWin.sessionStorage.length; i++) {
                const k = iframeWin.sessionStorage.key(i);
                session[k] = iframeWin.sessionStorage.getItem(k);
            }
            const savePayload = {
                game: title,
                savedAt: new Date().toISOString(),
                localStorage: data,
                sessionStorage: session
            };
            const blob = new Blob([JSON.stringify(savePayload, null, 2)], { type: 'application/json' });
            const a = document.createElement('a');
            a.href = URL.createObjectURL(blob);
            a.download = `${(title || 'game').replace(/\s+/g, '-').toLowerCase()}-save.json`;
            a.click();
            setTimeout(() => URL.revokeObjectURL(a.href), 1000);
            showGameToast('💾 Save downloaded!');
        } catch (e) {
            showGameToast('⚠ Could not read game data (cross-origin restriction)', true);
        }
    });

    // ---- LOAD: import a .json save back into iframe localStorage ----
    viewer.querySelector('#gv-load').addEventListener('click', () => {
        const fileInput = document.createElement('input');
        fileInput.type = 'file';
        fileInput.accept = '.json';
        fileInput.addEventListener('change', () => {
            const file = fileInput.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const payload = JSON.parse(e.target.result);
                    const iframeWin = iframe.contentWindow;
                    // Restore localStorage
                    if (payload.localStorage) {
                        Object.entries(payload.localStorage).forEach(([k, v]) => {
                            iframeWin.localStorage.setItem(k, v);
                        });
                    }
                    // Restore sessionStorage
                    if (payload.sessionStorage) {
                        Object.entries(payload.sessionStorage).forEach(([k, v]) => {
                            iframeWin.sessionStorage.setItem(k, v);
                        });
                    }
                    showGameToast('✓ Save loaded! Refreshing...');
                    // Reload the iframe so the game picks up the new storage
                    setTimeout(() => {
                        const src = iframe.src;
                        iframe.style.opacity = '0';
                        loadingEl.style.display = 'flex';
                        iframe.src = '';
                        setTimeout(() => {
                            iframe.src = src;
                            iframe.addEventListener('load', () => {
                                loadingEl.style.display = 'none';
                                iframe.style.opacity = '1';
                            }, { once: true });
                        }, 80);
                    }, 800);
                } catch {
                    showGameToast('⚠ Invalid save file', true);
                }
            };
            reader.readAsText(file);
        });
        fileInput.click();
    });

    // ---- REFRESH ----
    viewer.querySelector('#gv-refresh').addEventListener('click', () => {
        const src = iframe.src;
        iframe.style.opacity = '0';
        loadingEl.style.display = 'flex';
        iframe.src = '';
        setTimeout(() => {
            iframe.src = src;
            iframe.addEventListener('load', () => {
                loadingEl.style.display = 'none';
                iframe.style.opacity = '1';
            }, { once: true });
        }, 80);
    });

    // ---- FULLSCREEN ----
    viewer.querySelector('#gv-fullscreen').addEventListener('click', () => {
        const wrap = viewer.querySelector('.game-iframe-wrap');
        if (document.fullscreenElement) document.exitFullscreen();
        else (wrap.requestFullscreen || wrap.webkitRequestFullscreen || function(){}).call(wrap);
    });

    // ---- EXIT ----
    viewer.querySelector('#gv-exit').addEventListener('click', () => closeGameViewer(viewer));

    // ESC key
    const escHandler = (e) => {
        if (e.key === 'Escape' && !document.fullscreenElement) {
            closeGameViewer(viewer);
            document.removeEventListener('keydown', escHandler);
        }
    };
    document.addEventListener('keydown', escHandler);
    viewer._escHandler = escHandler;
}

function closeGameViewer(viewer, animate = true) {
    if (!viewer || !viewer.isConnected) return;
    if (viewer._escHandler) document.removeEventListener('keydown', viewer._escHandler);
    if (animate) {
        viewer.classList.add('closing');
        setTimeout(() => { viewer.remove(); if (currentGameViewer === viewer) currentGameViewer = null; }, 200);
    } else {
        viewer.remove();
        if (currentGameViewer === viewer) currentGameViewer = null;
    }
}

    document.body.appendChild(t);
    setTimeout(() => { t.classList.add('out'); setTimeout(() => t.remove(), 200); }, 2500);
}

const SAVE_KEY = 'fluxus_saves';
function escHtml(s) { return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }
function loadSaves() { try { return JSON.parse(localStorage.getItem(SAVE_KEY)||'[]'); } catch { return []; } }
function persistSaves(s) { localStorage.setItem(SAVE_KEY, JSON.stringify(s)); }
function formatDate(ts) {
    const d = new Date(ts);
    return d.toLocaleDateString(undefined, {month:'short',day:'numeric',year:'numeric'}) + ' · ' + d.toLocaleTimeString(undefined, {hour:'2-digit',minute:'2-digit'});
}

    if (isError){t.style.background='#ff4444';t.style.color='#fff';}
    document.body.appendChild(t);
    setTimeout(()=>{t.classList.add('out');setTimeout(()=>t.remove(),250);},2200);
}

    const a=document.createElement('a'); a.href=URL.createObjectURL(b); a.download=filename; a.click();
    setTimeout(()=>URL.revokeObjectURL(a.href),1000);
}

    function importFromFile(file) {
        const r=new FileReader();
        r.onload=e=>{
            try {
                const imp=JSON.parse(e.target.result);
                if(!Array.isArray(imp))throw new Error();
                const ex=loadSaves(); const ids=new Set(ex.map(s=>s.id));

                render();
            } catch { showSaveToast('⚠ Invalid save file',true); }
        };
        r.readAsText(file);
    }

    document.body.appendChild(modal);
    render();
}
