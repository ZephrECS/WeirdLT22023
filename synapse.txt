/* ════ LAUNCH GAME — spinner overlay → new window ════ */
async function launchGame(url,title,loadMode,gameIdx){
  // Show blurred overlay with spinner for 0.8s
  const ov=document.createElement('div');ov.className='game-launch-ov';
  ov.innerHTML=`<div class="launch-spinner-ring"></div><div class="launch-title">${title||'Loading...'}</div><div class="launch-sub">Preparing game</div>`;
  document.body.appendChild(ov);

  try{
    const res=await fetch(url);
    const code=await res.text();
    // Wait at least 0.8s from when overlay appeared
    await new Promise(r=>setTimeout(r,800));
    // Dismiss overlay
    ov.classList.add('out');setTimeout(()=>ov.remove(),280);
    // Open in new window
    const tab=window.open('','_blank');
    if(tab){tab.document.open();tab.document.write(code);tab.document.close();}
    else{alert('Popup blocked! Please allow popups for this site.');}
  }catch{
    ov.classList.add('out');setTimeout(()=>ov.remove(),280);
    showToast('⚠ Failed to load game');
  }
}

/* ════ VISITOR COUNTER — observe DOM for injected number ════ */
function watchForCounter(){
  const raw=document.getElementById('vc-raw');if(!raw)return;
  const mo=new MutationObserver(()=>{
    const num=raw.querySelector('span,div,td,b,strong');
    if(num&&num.textContent.trim().match(/\d+/)){
      document.getElementById('vc-num').textContent=num.textContent.trim().replace(/\D/g,'').replace(/\B(?=(\d{3})+(?!\d))/g,',');
      mo.disconnect();
    }
  });
  mo.observe(raw,{childList:true,subtree:true,characterData:true});
  // Also try after delays as fallback
  [1500,3000,5000].forEach(d=>setTimeout(()=>{
    const spans=raw.querySelectorAll('*');
    for(const s of spans){if(s.textContent.trim().match(/^\d+$/)){document.getElementById('vc-num').textContent=s.textContent.trim().replace(/\B(?=(\d{3})+(?!\d))/g,',');mo.disconnect();return;}}
  },d));
}
