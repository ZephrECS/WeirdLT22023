/* ── SNOW ─────────────────────────────────────────────── */
const canvas=document.getElementById('snow'),ctx=canvas.getContext('2d');
let flakes=[];
function initSnow(){
  canvas.width=window.innerWidth;canvas.height=window.innerHeight;
  flakes=Array.from({length:130},()=>{
    const r=Math.random();
    return{x:Math.random()*canvas.width,y:Math.random()*canvas.height,
      rad:.4+Math.pow(r,2)*2.4,op:.03+r*.16,sp:.12+r*.36,
      dr:(Math.random()-.5)*.28,wb:Math.random()*Math.PI*2};
  });
}
function drawSnow(){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  flakes.forEach(f=>{
    f.y+=f.sp;f.wb+=.008;f.x+=Math.sin(f.wb)*f.dr;
    if(f.y>canvas.height){f.y=-4;f.x=Math.random()*canvas.width;}
    ctx.beginPath();ctx.arc(f.x,f.y,f.rad,0,Math.PI*2);
    ctx.fillStyle=`rgba(255,255,255,${f.op})`;ctx.fill();
  });
  requestAnimationFrame(drawSnow);
}
window.addEventListener('resize',initSnow);
initSnow();drawSnow();

    el.appendChild(d);
  }
}

function loadData(){
  const input=document.createElement('input');input.type='file';input.accept='.json';
  input.onchange=e=>{
    const f=e.target.files[0];if(!f)return;
    const r=new FileReader();
    r.onload=ev=>{
      try{
        Object.assign(state,JSON.parse(ev.target.result));
        persist();applyTheme(state.theme);applyCloak(state.cloak);
        document.getElementById('panicDisp').textContent=state.panicKey;
        document.getElementById('panicUrlInput').value=state.panicUrl;
        if(state.experience==='full')document.getElementById('navProxy').style.display='';
        state.onboarded=true;persist();
        document.getElementById('onboarding').classList.add('hide');
        renderAll();showToast('Data loaded!');
      }catch(e){showToast('Invalid file');}
    };
    r.readAsText(f);
  };
  input.click();
}

function goProxy(){
  let url=document.getElementById('proxyInput').value.trim();if(!url)return;
  if(!url.startsWith('http'))url='https://'+url;
  document.getElementById('proxyWrap').innerHTML=`<iframe src="${url}" style="width:100%;height:100%;border:none;" sandbox="allow-scripts allow-same-origin allow-forms"></iframe>`;
}
function homeSearchKey(e){
  if(e.key!=='Enter')return;
  const v=document.getElementById('homeSearch').value.trim();if(!v)return;
  const url=v.includes('.')&&!v.includes(' ')?(v.startsWith('http')?v:'https://'+v):`https://www.google.com/search?q=${encodeURIComponent(v)}`;
  if(state.experience==='full'){switchTab('proxy');document.getElementById('proxyInput').value=url;goProxy();}
  else window.open(url,'_blank');
}
