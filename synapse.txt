/* LOADER */
(function(){
  const bar=document.getElementById('lbar'),status=document.getElementById('lstatus');
  const msgs=['loading games','preloading assets','finalizing','ready'];
  let p=0,mi=0,ti=0;
  const targets=[25,60,88,100];
  function tick(){
    if(ti>=targets.length){launch();return;}
    if(p<targets[ti]){
      p++; bar.style.width=p+'%';
      if(p===25||p===60||p===90||p===100){if(mi<msgs.length)status.textContent=msgs[mi++];}
      setTimeout(tick,12+Math.random()*24);
    } else {ti++;setTimeout(tick,200);}
  }
  function launch(){
    setTimeout(()=>{
      document.getElementById('loader').classList.add('done');
      document.getElementById('app').classList.add('visible');
      init();
    },350);
  }
  setTimeout(tick,300);
})();

/* STATE */
let activeViewer=null, heroGames=[], heroIdx=0, heroTimer=null;

function init(){
  heroGames=GAMES.filter(g=>g.cat==='featured');
  buildHero(0);
  buildHeroDots();
  buildRows();
  bindNavCats();
  document.getElementById('searchInput').addEventListener('input',function(){
    const v=this.value.trim();
    if(!v)showHome(); else showSearch(v);
  });
  window.addEventListener('scroll',()=>{
    document.getElementById('nav').classList.toggle('solid',window.scrollY>20);
  });
}

function buildHero(idx){
  heroIdx=idx;
  const g=heroGames[idx]; if(!g)return;
  document.getElementById('heroBg').style.backgroundImage=`url('${g.img}')`;
  document.getElementById('heroTitle').textContent=g.title;
  document.getElementById('heroBadge').textContent=g.cat.charAt(0).toUpperCase()+g.cat.slice(1);
  document.getElementById('heroPlay').onclick=()=>{ if(g.versions)versionPicker(g); else launchGame(g.url,g.title,g.mode||'onsite'); };
  document.querySelectorAll('.hero-dot').forEach((d,i)=>d.classList.toggle('active',i===idx));
  clearTimeout(heroTimer);
  heroTimer=setTimeout(()=>buildHero((idx+1)%heroGames.length),6000);
}
function buildHeroDots(){
  const w=document.getElementById('heroDots'); w.innerHTML='';
  heroGames.forEach((_,i)=>{
    const d=document.createElement('div'); d.className='hero-dot'+(i===0?' active':'');
    d.onclick=()=>buildHero(i); w.appendChild(d);
  });
}
function buildRows(){
  const c=document.getElementById('rowsContainer'); c.innerHTML='';
  [{label:'Featured',cat:'featured'},{label:'Action',cat:'action'},{label:'Horror',cat:'horror'},{label:'Sandbox',cat:'sandbox'},{label:'Classic',cat:'classic'}].forEach(({label,cat})=>{
    const games=GAMES.filter(g=>g.cat===cat); if(!games.length)return;
    const row=document.createElement('div'); row.className='row';
    row.innerHTML=`<div class="row-header"><div class="row-label">${label}</div><div class="row-see-all" data-cat="${cat}" data-label="${label}">See all →</div></div><div class="shelf" id="shelf-${cat}"></div>`;
    c.appendChild(row);
    row.querySelector('.row-see-all').onclick=e=>showCat(e.target.dataset.cat,e.target.dataset.label);
    const shelf=row.querySelector('.shelf');
    games.forEach((g,i)=>{const card=buildCard(g,false); card.style.animationDelay=(i*0.03)+'s'; shelf.appendChild(card);});
  });
}

function buildCard(g,isSecret){
  const d=document.createElement('div'); d.className='card';
  d.innerHTML=`
    <div class="card-img-wrap">
      <img class="card-img" src="${g.img}" alt="${g.title}" loading="lazy">
      <div class="card-overlay"><div class="card-play-btn"><svg viewBox="0 0 24 24"><path d="M8 5v14l11-7z"/></svg></div></div>
    </div>
    <div class="card-bottom"><div class="card-name">${g.title}</div><div class="card-cat">${g.cat||'secret'}</div></div>`;
  d.onclick=()=>{ if(g.versions)versionPicker(g); else launchGame(g.url,g.title,g.mode||'onsite'); };
  return d;
}

function setView(v){
  document.getElementById('homeView').style.display=v==='home'?'block':'none';
  document.getElementById('catView').classList.toggle('visible',v==='cat');
  document.getElementById('searchView').classList.toggle('visible',v==='search');
}
function showHome(){
  setView('home');
  document.querySelectorAll('.nav-cat').forEach(el=>el.classList.toggle('active',el.dataset.cat==='all'));
  document.getElementById('searchInput').value='';
  window.scrollTo(0,0);
}
function showCat(cat,label){
  const games=GAMES.filter(g=>g.cat===cat);
  const grid=document.getElementById('catGrid'); grid.innerHTML='';
  games.forEach((g,i)=>{const c=buildCard(g,false); c.style.animationDelay=(i*0.025)+'s'; grid.appendChild(c);});
  document.getElementById('catHeading').textContent=label+' — '+games.length+' games';
  setView('cat');
  document.querySelectorAll('.nav-cat').forEach(el=>el.classList.toggle('active',el.dataset.cat===cat));
  window.scrollTo(0,0);
}
function showSearch(q){
  if(!q){showHome();return;}
  let list=GAMES.filter(g=>g.title.toLowerCase().includes(q.toLowerCase()));
  const sg=SECRET.find(s=>s.trigger===q.toLowerCase().trim());
  if(sg)list=[sg,...list];
  const grid=document.getElementById('searchGrid'); grid.innerHTML='';
  if(!list.length){grid.innerHTML='<div class="no-results">No games found.</div>';}
  else{list.forEach((g,i)=>{const c=buildCard(g,!!g.trigger); c.style.animationDelay=(i*0.025)+'s'; grid.appendChild(c);});}
  document.getElementById('searchLabel').textContent=`"${q}" — ${list.length} result${list.length===1?'':'s'}`;
  setView('search');
  document.querySelectorAll('.nav-cat').forEach(el=>el.classList.remove('active'));
  window.scrollTo(0,0);
}
function bindNavCats(){
  document.querySelectorAll('.nav-cat').forEach(el=>{
    el.addEventListener('click',()=>{
      const cat=el.dataset.cat;
      if(cat==='all'){showHome();return;}
      const labels={featured:'Featured',action:'Action',horror:'Horror',sandbox:'Sandbox',classic:'Classic'};
      showCat(cat,labels[cat]||cat);
    });
  });
}

function versionPicker(game){
  const bg=document.createElement('div'); bg.className='modal-bg';
  bg.innerHTML=`<div class="modal"><div class="modal-title">${game.title}</div><div class="modal-sub">choose version</div><div class="modal-opts">${game.versions.map(v=>`<div class="modal-opt" data-url="${v.url}" data-mode="${game.mode||'onsite'}"><div><div class="modal-opt-name">${v.name}</div><div class="modal-opt-desc">${v.desc}</div></div><span class="modal-opt-arrow">›</span></div>`).join('')}</div><button class="modal-cancel">Cancel</button></div>`;
  const close=()=>{bg.classList.add('out');setTimeout(()=>bg.remove(),200);};
  bg.querySelectorAll('.modal-opt').forEach(o=>o.addEventListener('click',()=>{close();setTimeout(()=>launchGame(o.dataset.url,game.title,o.dataset.mode),180);}));
  bg.querySelector('.modal-cancel').addEventListener('click',close);
  bg.addEventListener('click',e=>{if(e.target===bg)close();});
  document.body.appendChild(bg);
}

async function launchGame(url,title,mode){
  if(mode==='newtab'){
    try{const r=await fetch(url);const h=await r.text();const t=window.open('','_blank');t.document.open();t.document.write(h);t.document.close();}
    catch{alert('Could not load game.');}
    return;
  }
  openViewer(url,title);
}

  v.querySelector('#vbLoad').onclick=()=>{
    const fi=document.createElement('input');fi.type='file';fi.accept='.json';
    fi.onchange=()=>{
      const f=fi.files[0];if(!f)return;
      const r=new FileReader();
      r.onload=e=>{
        try{
          const p=JSON.parse(e.target.result),iw=iframe.contentWindow;
          if(p.localStorage)Object.entries(p.localStorage).forEach(([k,val])=>iw.localStorage.setItem(k,val));
          if(p.sessionStorage)Object.entries(p.sessionStorage).forEach(([k,val])=>iw.sessionStorage.setItem(k,val));
          toast('Save loaded — restarting');
          setTimeout(()=>{const src=iframe.src;iframe.style.opacity='0';loader.style.display='flex';iframe.src='';setTimeout(()=>{iframe.src=src;iframe.addEventListener('load',()=>{loader.style.display='none';iframe.style.opacity='1';},{once:true});},80);},700);
        } catch{toast('Invalid save file',true);}
      };r.readAsText(f);
    };fi.click();
  };
  v.querySelector('#vbRefresh').onclick=()=>{const src=iframe.src;iframe.style.opacity='0';loader.style.display='flex';iframe.src='';setTimeout(()=>{iframe.src=src;iframe.addEventListener('load',()=>{loader.style.display='none';iframe.style.opacity='1';},{once:true});},80);};
  v.querySelector('#vbFS').onclick=()=>{const w=v.querySelector('.vframe');if(document.fullscreenElement)document.exitFullscreen();else(w.requestFullscreen||w.webkitRequestFullscreen||function(){}).call(w);};
  v.querySelector('#vbExit').onclick=()=>killViewer(v);
  const esc=e=>{if(e.key==='Escape'&&!document.fullscreenElement){killViewer(v);document.removeEventListener('keydown',esc);}};
  document.addEventListener('keydown',esc); v._esc=esc;
}

function killViewer(v,anim=true){
  if(!v||!v.isConnected)return;
  if(v._esc)document.removeEventListener('keydown',v._esc);
  if(anim){v.classList.add('out');setTimeout(()=>{v.remove();if(activeViewer===v)activeViewer=null;},160);}
  else{v.remove();if(activeViewer===v)activeViewer=null;}
}
function toast(msg,isErr=false){
  const ex=document.querySelector('.toast');if(ex)ex.remove();
  const t=document.createElement('div');t.className='toast'+(isErr?' err':'');t.textContent=msg;
  document.body.appendChild(t);
  setTimeout(()=>{t.classList.add('out');setTimeout(()=>t.remove(),200);},2600);
}
